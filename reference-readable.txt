    /***************************************************************************
     * 2. MOTOR ACÚSTICO FOLEY PROCEDURAL (WEB AUDIO API — CERO ARCHIVOS EXTERNOS)
     ***************************************************************************/
    class AntiqueFoleyAudio {
      constructor() {
        this.ctx = null;
        this.enabled = true;
        this.ambientStarted = false;
        this.masterGain = null;
        this.fireInterval = null;
      }

      initContext() {
        if (!this.ctx) {
          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          if (!AudioCtx) return;
          this.ctx = new AudioCtx();
          this.masterGain = this.ctx.createGain();
          this.masterGain.gain.value = this.enabled ? 0.85 : 0.0;
          this.masterGain.connect(this.ctx.destination);
        }
        if (this.ctx && this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
        if (this.ctx && !this.ambientStarted) {
          this.startCampfireAmbience();
        }
      }

      toggle() {
        this.initContext();
        this.enabled = !this.enabled;
        if (this.masterGain && this.ctx) {
          this.masterGain.gain.setTargetAtTime(this.enabled ? 0.85 : 0.0, this.ctx.currentTime, 0.08);
        }
        return this.enabled;
      }

      startCampfireAmbience() {
        if (!this.ctx || this.ambientStarted) return;
        this.ambientStarted = true;

        // Warm low rumble of the fire
        const bufferSize = this.ctx.sampleRate * 3;
        const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const output = noiseBuffer.getChannelData(0);
        let lastOut = 0.0;
        for (let i = 0; i < bufferSize; i++) {
          const white = Math.random() * 2 - 1;
          output[i] = (lastOut + 0.02 * white) / 1.02;
          lastOut = output[i];
          output[i] *= 1.8;
        }

        const brownNoise = this.ctx.createBufferSource();
        brownNoise.buffer = noiseBuffer;
        brownNoise.loop = true;

        const filter = this.ctx.createBiquadFilter();
        filter.type = 'lowpass';
        filter.frequency.value = 195;

        const fireGain = this.ctx.createGain();
        fireGain.gain.value = 0.14;

        brownNoise.connect(filter);
        filter.connect(fireGain);
        fireGain.connect(this.masterGain);
        brownNoise.start(0);

        // Stochastic wood crackles & pops
        if (this.fireInterval) clearInterval(this.fireInterval);
        this.fireInterval = setInterval(() => {
          if (!this.enabled || !this.ctx || this.ctx.state !== 'running') return;
          if (Math.random() < 0.68) {
            this.triggerCracklePop();
          }
        }, 320);
      }

      triggerCracklePop() {
        if (!this.ctx) return;
        const now = this.ctx.currentTime;
        const dur = 0.02 + Math.random() * 0.035;
        const buf = this.ctx.createBuffer(1, Math.floor(this.ctx.sampleRate * dur), this.ctx.sampleRate);
        const data = buf.getChannelData(0);
        for (let i = 0; i < data.length; i++) {
          data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (data.length * 0.3));
        }

        const src = this.ctx.createBufferSource();
        src.buffer = buf;

        const bpf = this.ctx.createBiquadFilter();
        bpf.type = 'bandpass';
        bpf.frequency.value = 1400 + Math.random() * 2400;
        bpf.Q.value = 2.2;

        const gain = this.ctx.createGain();
        gain.gain.value = 0.04 + Math.random() * 0.07;

        src.connect(bpf);
        bpf.connect(gain);
        gain.connect(this.masterGain);
        src.start(now);
      }

      // Heavy leather book grab & lift thud + leather creak
      playHeavyBookGrab() {
        this.initContext();
        if (!this.ctx || !this.enabled) return;
        const now = this.ctx.currentTime;

        // Low heavy wooden/leather weight thud
        const osc = this.ctx.createOscillator();
        const thudGain = this.ctx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(112, now);
        osc.frequency.exponentialRampToValueAtTime(38, now + 0.28);

        thudGain.gain.setValueAtTime(0.32, now);
        thudGain.gain.exponentialRampToValueAtTime(0.001, now + 0.32);

        osc.connect(thudGain);
        thudGain.connect(this.masterGain);
        osc.start(now);
        osc.stop(now + 0.34);

        // Leather creak burst
        setTimeout(() => this.playLeatherCreak(0.38), 180);
      }

      playLeatherCreak(duration = 0.45) {
        this.initContext();
        if (!this.ctx || !this.enabled) return;
        const now = this.ctx.currentTime;
        const samples = Math.floor(this.ctx.sampleRate * duration);
        const buf = this.ctx.createBuffer(1, samples, this.ctx.sampleRate);
        const data = buf.getChannelData(0);
        for (let i = 0; i < samples; i++) {
          const t = i / samples;
          const grain = Math.sin(i * 0.045 + Math.sin(i * 0.008) * 6);
          data[i] = ((Math.random() * 2 - 1) * 0.4 + grain * 0.6) * Math.sin(t * Math.PI);
        }

        const src = this.ctx.createBufferSource();
        src.buffer = buf;

        const filter = this.ctx.createBiquadFilter();
        filter.type = 'bandpass';
        filter.frequency.setValueAtTime(340, now);
        filter.frequency.linearRampToValueAtTime(520, now + duration * 0.6);
        filter.Q.value = 3.5;

        const gain = this.ctx.createGain();
        gain.gain.value = 0.12;

        src.connect(filter);
        filter.connect(gain);
        gain.connect(this.masterGain);
        src.start(now);
      }

      // Thick parchment sheet turn rustle
      playPageTurn() {
        this.initContext();
        if (!this.ctx || !this.enabled) return;
        const now = this.ctx.currentTime;
        const dur = 0.36;
        const samples = Math.floor(this.ctx.sampleRate * dur);
        const buf = this.ctx.createBuffer(1, samples, this.ctx.sampleRate);
        const data = buf.getChannelData(0);
        for (let i = 0; i < samples; i++) {
          const t = i / samples;
          const env = Math.pow(Math.sin(t * Math.PI), 1.4);
          data[i] = (Math.random() * 2 - 1) * env;
        }

        const src = this.ctx.createBufferSource();
        src.buffer = buf;

        const filter = this.ctx.createBiquadFilter();
        filter.type = 'bandpass';
        filter.frequency.setValueAtTime(900, now);
        filter.frequency.exponentialRampToValueAtTime(2200, now + dur * 0.5);
        filter.frequency.exponentialRampToValueAtTime(750, now + dur);
        filter.Q.value = 1.4;

        const gain = this.ctx.createGain();
        gain.gain.value = 0.16;

        src.connect(filter);
        filter.connect(gain);
        gain.connect(this.masterGain);
        src.start(now);
      }

      playWaxStamp() {
        this.initContext();
        if (!this.ctx || !this.enabled) return;
        this.playHeavyBookGrab();
      }
    }

    function bakeFrontCoverPBR(portraitImgElement) {
      const w = 1024;
      const h = 1365;

      const albedoCanvas = document.createElement('canvas');
      albedoCanvas.width = w;
      albedoCanvas.height = h;
      const ctx = albedoCanvas.getContext('2d');

      const bumpCanvas = document.createElement('canvas');
      bumpCanvas.width = w;
      bumpCanvas.height = h;
      const bCtx = bumpCanvas.getContext('2d');

      // 1. Base worn cordovan brown leather gradient
      const baseGrad = ctx.createRadialGradient(w * 0.5, h * 0.48, w * 0.1, w * 0.5, h * 0.5, w * 0.85);
      baseGrad.addColorStop(0, '#4c2712');
      baseGrad.addColorStop(0.55, '#34190a');
      baseGrad.addColorStop(1, '#190b04');
      ctx.fillStyle = baseGrad;
      ctx.fillRect(0, 0, w, h);

      // Base neutral-mid bump
      bCtx.fillStyle = '#787878';
      bCtx.fillRect(0, 0, w, h);

      // 2. Leather grain pores & aged scratches
      for (let i = 0; i < 6500; i++) {
        const px = Math.random() * w;
        const py = Math.random() * h;
        const r = 0.8 + Math.random() * 2.6;
        ctx.fillStyle = i % 3 === 0 ? 'rgba(95, 50, 22, 0.16)' : 'rgba(15, 7, 2, 0.24)';
        ctx.beginPath();
        ctx.arc(px, py, r, 0, Math.PI * 2);
        ctx.fill();

        bCtx.fillStyle = i % 3 === 0 ? 'rgba(165, 165, 165, 0.15)' : 'rgba(65, 65, 65, 0.22)';
        bCtx.beginPath();
        bCtx.arc(px, py, r, 0, Math.PI * 2);
        bCtx.fill();
      }

      // Deep leather creases / cracks
      for (let c = 0; c < 110; c++) {
        let cx = Math.random() * w;
        let cy = Math.random() * h;
        ctx.strokeStyle = 'rgba(14, 6, 2, 0.42)';
        ctx.lineWidth = 0.8 + Math.random() * 1.8;
        bCtx.strokeStyle = 'rgba(35, 35, 35, 0.5)';
        bCtx.lineWidth = ctx.lineWidth + 0.6;

        ctx.beginPath();
        bCtx.beginPath();
        ctx.moveTo(cx, cy);
        bCtx.moveTo(cx, cy);
        const steps = 6 + Math.floor(Math.random() * 10);
        for (let s = 0; s < steps; s++) {
          cx += (Math.random() - 0.5) * 38;
          cy += (Math.random() - 0.5) * 38;
          ctx.lineTo(cx, cy);
          bCtx.lineTo(cx, cy);
        }
        ctx.stroke();
        bCtx.stroke();
      }

      // 3. Integrate the Author Portrait ("Kevin Stiven Buesaquillo Timarán") hyperrealistically onto the cover
      const frameX = 112;
      const frameY = 205;
      const frameW = w - frameX * 2;
      const frameH = 855;

      // Sunken blind-tooled recess in the leather for the portrait plate
      ctx.fillStyle = '#220f06';
      ctx.fillRect(frameX - 14, frameY - 14, frameW + 28, frameH + 28);
      bCtx.fillStyle = '#484848';
      bCtx.fillRect(frameX - 14, frameY - 14, frameW + 28, frameH + 28);

      if (portraitImgElement && portraitImgElement.complete && portraitImgElement.naturalWidth > 0) {
        // Draw portrait into an offscreen canvas to extract sepia leather-engraving & bump height
        const off = document.createElement('canvas');
        off.width = frameW;
        off.height = frameH;
        const oCtx = off.getContext('2d');

        // Cover-fit crop
        const imgAspect = portraitImgElement.naturalWidth / portraitImgElement.naturalHeight;
        const boxAspect = frameW / frameH;
        let sx = 0, sy = 0, sw = portraitImgElement.naturalWidth, sh = portraitImgElement.naturalHeight;
        if (imgAspect > boxAspect) {
          sw = portraitImgElement.naturalHeight * boxAspect;
          sx = (portraitImgElement.naturalWidth - sw) * 0.5;
        } else {
          sh = portraitImgElement.naturalWidth / boxAspect;
          sy = (portraitImgElement.naturalHeight - sh) * 0.5;
        }
        oCtx.drawImage(portraitImgElement, sx, sy, sw, sh, 0, 0, frameW, frameH);

        const imgData = oCtx.getImageData(0, 0, frameW, frameH);
        const d = imgData.data;
        const bumpData = oCtx.createImageData(frameW, frameH);
        const bd = bumpData.data;

        for (let p = 0; p < d.length; p += 4) {
          const r = d[p];
          const g = d[p + 1];
          const b = d[p + 2];
          // Luminance (strictly zero blue in final leather embossing!)
          let lum = (0.35 * r + 0.53 * g + 0.12 * b) / 255;
          // Increase contrast for intaglio leather engraving feel
          lum = Math.max(0, Math.min(1, (lum - 0.5) * 1.28 + 0.5));

          // Map luminance to warm espresso/sepia/ochre leather ink ramp
          const lr = Math.floor(28 + lum * 178);
          const lg = Math.floor(14 + lum * 122);
          const lb = Math.floor(5 + lum * 62);

          d[p] = lr;
          d[p + 1] = lg;
          d[p + 2] = lb;
          d[p + 3] = 242;

          // Bump height from portrait luminance so firelight catches the author's face & filigree!
          const bh = Math.floor(70 + lum * 135);
          bd[p] = bh;
          bd[p + 1] = bh;
          bd[p + 2] = bh;
          bd[p + 3] = 255;
        }

        oCtx.putImageData(imgData, 0, 0);
        ctx.save();
        ctx.drawImage(off, frameX, frameY);

        // Overlay subtle leather cracks across the portrait so it feels physically printed into the hide
        ctx.globalCompositeOperation = 'multiply';
        const vig = ctx.createRadialGradient(
          frameX + frameW * 0.5, frameY + frameH * 0.48, frameW * 0.22,
          frameX + frameW * 0.5, frameY + frameH * 0.5, frameW * 0.72
        );
        vig.addColorStop(0, 'rgba(255, 238, 210, 1)');
        vig.addColorStop(0.78, 'rgba(210, 165, 115, 0.92)');
        vig.addColorStop(1, 'rgba(75, 38, 16, 0.88)');
        ctx.fillStyle = vig;
        ctx.fillRect(frameX, frameY, frameW, frameH);
        ctx.restore();

        // Put portrait relief into bump map
        const bOff = document.createElement('canvas');
        bOff.width = frameW;
        bOff.height = frameH;
        bOff.getContext('2d').putImageData(bumpData, 0, 0);
        bCtx.drawImage(bOff, frameX, frameY);
      }

      // 4. Antique Gold-Leaf Embossed Borders & Classical Typography on the Cover
      ctx.strokeStyle = '#c6933c';
      ctx.lineWidth = 6;
      ctx.strokeRect(48, 48, w - 96, h - 96);
      ctx.lineWidth = 2;
      ctx.strokeRect(64, 64, w - 128, h - 128);
      ctx.strokeRect(frameX - 6, frameY - 6, frameW + 12, frameH + 12);

      bCtx.strokeStyle = '#e8e8e8';
      bCtx.lineWidth = 7;
      bCtx.strokeRect(48, 48, w - 96, h - 96);
      bCtx.lineWidth = 3;
      bCtx.strokeRect(64, 64, w - 128, h - 128);
      bCtx.strokeRect(frameX - 6, frameY - 6, frameW + 12, frameH + 12);

      // Top Embossed Title on the Leather Cover
      ctx.fillStyle = '#d4a249';
      ctx.textAlign = 'center';
      ctx.font = 'bold 34px "Cinzel", Georgia, serif';
      ctx.fillText('· CODEX OPERUM ·', w / 2, 122);
      ctx.font = 'italic 22px "EB Garamond", Georgia, serif';
      ctx.fillStyle = '#b88532';
      ctx.fillText('ARS ARCHITECTURA ET INGENIUM', w / 2, 164);

      bCtx.fillStyle = '#efefef';
      bCtx.textAlign = 'center';
      bCtx.font = 'bold 34px "Cinzel", Georgia, serif';
      bCtx.fillText('· CODEX OPERUM ·', w / 2, 122);

      // Bottom Embossed Author Name Plaque on the Leather Cover
      ctx.fillStyle = '#200e05';
      ctx.fillRect(112, 1105, w - 224, 168);
      ctx.strokeStyle = '#c6933c';
      ctx.lineWidth = 3;
      ctx.strokeRect(112, 1105, w - 224, 168);

      bCtx.fillStyle = '#555555';
      bCtx.fillRect(112, 1105, w - 224, 168);
      bCtx.strokeStyle = '#e5e5e5';
      bCtx.lineWidth = 4;
      bCtx.strokeRect(112, 1105, w - 224, 168);

      ctx.fillStyle = '#e0b158';
      ctx.font = 'bold 31px "Cinzel", Georgia, serif';
      ctx.fillText('KEVIN STIVEN', w / 2, 1162);
      ctx.font = 'bold 29px "Cinzel", Georgia, serif';
      ctx.fillText('BUESAQUILLO TIMARÁN', w / 2, 1204);
      ctx.fillStyle = '#b88738';
      ctx.font = 'italic 21px "EB Garamond", Georgia, serif';
      ctx.fillText('Edición Príncipe · Año de MMXXVI', w / 2, 1246);

      bCtx.fillStyle = '#f5f5f5';
      bCtx.font = 'bold 31px "Cinzel", Georgia, serif';
      bCtx.fillText('KEVIN STIVEN', w / 2, 1162);
      bCtx.font = 'bold 29px "Cinzel", Georgia, serif';
      bCtx.fillText('BUESAQUILLO TIMARÁN', w / 2, 1204);

      if (!pbrTextures.coverAlbedo) {
        pbrTextures.coverAlbedo = new THREE.CanvasTexture(albedoCanvas);
        pbrTextures.coverBump = new THREE.CanvasTexture(bumpCanvas);
      } else {
        pbrTextures.coverAlbedo.image = albedoCanvas;
        pbrTextures.coverBump.image = bumpCanvas;
      }
      pbrTextures.coverAlbedo.needsUpdate = true;
      pbrTextures.coverBump.needsUpdate = true;

      if (pbrTextures.coverMaterial) {
        pbrTextures.coverMaterial.map = pbrTextures.coverAlbedo;
        pbrTextures.coverMaterial.bumpMap = pbrTextures.coverBump;
        pbrTextures.coverMaterial.needsUpdate = true;
      }
    }

    // --- TRANSICIÓN 1 -> 2: Extender las manos, agarrar el pesado libro y sostenerlo cerrado ---
    function triggerGrabBookSequence() {
      if (currentState !== STATE.IDLE) return;
      currentState = STATE.GRABBING;
      foley.playHeavyBookGrab();

      if (activeTimeline) activeTimeline.kill();

      // Target held position directly in front of the avatar's chest/eyes
      const heldPos = { x: 0.0, y: 1.07, z: 0.93 };
      const heldRot = { x: -0.12, y: 0.0, z: 0.0 };

      activeTimeline = gsap.timeline({
        onComplete: () => {
          currentState = STATE.HELD_CLOSED;
          updateHUDForState(STATE.HELD_CLOSED);
        }
      });

      // 1. Smoothly center head gaze toward the campfire / held position
      activeTimeline.to(CAM_LOOK_BASE, {
        yaw: 0.0,
        pitch: -0.11,
        duration: 1.25,
        ease: 'power2.inOut'
      }, 0);

      // 2. Extend both 3D hands toward the resting tome on the right stump
      activeTimeline.to(leftHand.position, {
        x: BOOK_REST_POS.x - 0.16,
        y: BOOK_REST_POS.y - 0.02,
        z: BOOK_REST_POS.z + 0.06,
        duration: 0.58,
        ease: 'power2.out'
      }, 0);
      activeTimeline.to(rightHand.position, {
        x: BOOK_REST_POS.x + 0.16,
        y: BOOK_REST_POS.y - 0.02,
        z: BOOK_REST_POS.z - 0.02,
        duration: 0.58,
        ease: 'power2.out'
      }, 0);

      // 3. Lift the heavy tome (with a slight gravitational dip at the start to convey physical mass!)
      activeTimeline.to(bookRoot.position, {
        keyframes: [
          { x: 0.34, y: 0.49, z: 0.82, duration: 0.24, ease: 'power1.in' }, // physical weight dip
          { x: heldPos.x, y: heldPos.y, z: heldPos.z, duration: 0.88, ease: 'power3.out' }
        ]
      }, 0.52);

      activeTimeline.to(bookRoot.rotation, {
        x: heldRot.x,
        y: heldRot.y,
        z: heldRot.z,
        duration: 1.1,
        ease: 'power3.inOut'
      }, 0.52);

      // 4. Hands cradle both sides of the closed tome in front of the camera
      activeTimeline.to(leftHand.position, {
        x: heldPos.x - BOOK_W * 0.54,
        y: heldPos.y - 0.03,
        z: heldPos.z + 0.02,
        duration: 1.1,
        ease: 'power3.inOut'
      }, 0.52);
      activeTimeline.to(leftHand.rotation, {
        x: 0.22,
        y: 0.38,
        z: -0.48,
        duration: 1.1,
        ease: 'power3.inOut'
      }, 0.52);

      activeTimeline.to(rightHand.position, {
        x: heldPos.x + BOOK_W * 0.54,
        y: heldPos.y - 0.03,
        z: heldPos.z + 0.02,
        duration: 1.1,
        ease: 'power3.inOut'
      }, 0.52);
      activeTimeline.to(rightHand.rotation, {
        x: 0.22,
        y: -0.38,
        z: 0.48,
        duration: 1.1,
        ease: 'power3.inOut'
      }, 0.52);
    }

    // --- TRANSICIÓN 2 -> 3 -> 4: Abrir la pesada tapa de cuero + Zoom Ease-in-out hacia la Página Única ---
    function triggerOpenBookAndZoom() {
      if (currentState !== STATE.HELD_CLOSED) return;
      currentState = STATE.OPENING_ZOOM;
      foley.playLeatherCreak(0.65);
      setTimeout(() => foley.playPageTurn(), 420);

      if (activeTimeline) activeTimeline.kill();

      activeTimeline = gsap.timeline({
        onComplete: () => {
          currentState = STATE.READING_SINGLE_PAGE;
          updateHUDForState(STATE.READING_SINGLE_PAGE);
          singlePageReader.classList.add('active');
          singlePageReader.setAttribute('aria-hidden', 'false');
          // Hide 3D canvas completely so 100% of attention & viewport is the single parchment leaf
          container.style.opacity = '0';
          container.style.pointerEvents = 'none';
          renderActiveFolio(currentFolioIndex, false);
        }
      });

      // Straighten the book face-on toward the viewer's eyes
      activeTimeline.to(bookRoot.rotation, {
        x: 0.0,
        y: 0.0,
        z: 0.0,
        duration: 0.55,
        ease: 'power2.out'
      }, 0);

      // Swing open the heavy front cover on its left hinge (-152 degrees)
      activeTimeline.to(frontCoverPivot.rotation, {
        y: -2.65,
        duration: 1.25,
        ease: 'power3.inOut'
      }, 0.1);

      // Left hand shifts outward with the opened cover
      activeTimeline.to(leftHand.position, {
        x: -0.34,
        z: 0.98,
        duration: 1.15,
        ease: 'power3.inOut'
      }, 0.1);

      // Ease-in-out Camera Zoom directly into the single right-hand parchment leaf!
      activeTimeline.to(camera.position, {
        x: -0.005,
        y: 1.07,
        z: 1.055,
        duration: 1.45,
        ease: 'power3.inOut'
      }, 0.25);

      activeTimeline.to(CAM_LOOK_BASE, {
        yaw: 0.0,
        pitch: 0.0,
        duration: 1.25,
        ease: 'power3.inOut'
      }, 0.25);

      // Cross-fade into the full-viewport Single Parchment Leaf at the climax of the zoom
      activeTimeline.to(singlePageReader, {
        opacity: 1,
        duration: 0.55,
        ease: 'power2.inOut',
        onStart: () => {
          singlePageReader.classList.add('active');
        }
      }, 1.22);
    }

    // --- TRANSICIÓN 4 -> 2: Cerrar el Códice y regresar suavemente al entorno 3D frente a la fogata ---
    function triggerCloseBookToHands() {
      if (currentState !== STATE.READING_SINGLE_PAGE) return;
      currentState = STATE.OPENING_ZOOM;
      foley.playPageTurn();
      setTimeout(() => foley.playLeatherCreak(0.45), 260);

      container.style.opacity = '1';
      container.style.pointerEvents = 'auto';

      if (activeTimeline) activeTimeline.kill();

      activeTimeline = gsap.timeline({
        onComplete: () => {
          singlePageReader.classList.remove('active');
          singlePageReader.setAttribute('aria-hidden', 'true');
          currentState = STATE.HELD_CLOSED;
          updateHUDForState(STATE.HELD_CLOSED);
        }
      });

      activeTimeline.to(singlePageReader, {
        opacity: 0,
        duration: 0.45,
        ease: 'power2.out'
      }, 0);

      // Pull camera back to seated avatar position
      activeTimeline.to(camera.position, {
        x: CAM_IDLE_POS.x,
        y: CAM_IDLE_POS.y,
        z: CAM_IDLE_POS.z,
        duration: 1.25,
        ease: 'power3.inOut'
      }, 0.1);

      activeTimeline.to(CAM_LOOK_BASE, {
        yaw: 0.0,
        pitch: -0.11,
        duration: 1.25,
        ease: 'power3.inOut'
      }, 0.1);

      activeTimeline.to(frontCoverPivot.rotation, {
        y: 0.0,
        duration: 1.15,
        ease: 'power3.inOut'

      activeTimeline.to(bookRoot.rotation, {
        x: -0.12,
        y: 0.0,
        z: 0.0,

        ease: 'power3.inOut'

        ease: 'power3.inOut'
      }, 0.22);

    window.addEventListener('mousemove', (e) => {
      pointerNorm.targetX = (e.clientX / window.innerWidth) * 2 - 1;
      pointerNorm.targetY = (e.clientY / window.innerHeight) * 2 - 1;

      // Update warm candle/torch glow position on the single parchment page in Fase 4
      if (currentState === STATE.READING_SINGLE_PAGE) {
        const px = Math.round((e.clientX / window.innerWidth) * 100);
        const py = Math.round((e.clientY / window.innerHeight) * 100);
        parchmentSheet.style.setProperty('--torch-x', px + '%');
        parchmentSheet.style.setProperty('--torch-y', py + '%');
        return;
      }

      // Raycast against the 3D book in IDLE or HELD_CLOSED
      mouseVec.x = pointerNorm.targetX;
      mouseVec.y = -pointerNorm.targetY;
      raycaster.setFromCamera(mouseVec, camera);
      const hits = raycaster.intersectObjects(bookHitMeshes, false);
      isHoveringBook = hits.length > 0;
      container.classList.toggle('pointer-book', isHoveringBook);
    });

    // Touch support for mobile/tablet head-look
    window.addEventListener('touchmove', (e) => {
      if (!e.touches || e.touches.length === 0) return;
      const t = e.touches[0];
      pointerNorm.targetX = (t.clientX / window.innerWidth) * 2 - 1;
      pointerNorm.targetY = (t.clientY / window.innerHeight) * 2 - 1;
    }, { passive: true });

    // Clicking the 3D book in the scene triggers the appropriate phase transition
    container.addEventListener('click', (e) => {
      foley.initContext();
      mouseVec.x = (e.clientX / window.innerWidth) * 2 - 1;
      mouseVec.y = -(e.clientY / window.innerHeight) * 2 + 1;
      raycaster.setFromCamera(mouseVec, camera);
      const hits = raycaster.intersectObjects(bookHitMeshes, false);

      if (hits.length > 0) {
        if (currentState === STATE.IDLE) {
          triggerGrabBookSequence();
        } else if (currentState === STATE.HELD_CLOSED) {
          triggerOpenBookAndZoom();
        }
      }
    });

    function renderActiveFolio(targetIndex, withPageTurnAnim = true) {
      const clamped = Math.max(0, Math.min(FOLIO_METADATA.length - 1, targetIndex));
      currentFolioIndex = clamped;

      const applyDOMSwitch = () => {
        folioPages.forEach((sec, idx) => {
          sec.classList.toggle('hidden', idx !== clamped);
        });

        folioTabBtns.forEach((btn, idx) => {
          if (idx === clamped) {
            btn.className = 'folio-tab-btn px-2.5 py-1 text-xs font-cinzel tracking-wider border border-[#5c3818] bg-[#4a260c] text-[#f3e3c2]';
          } else {
            btn.className = 'folio-tab-btn px-2.5 py-1 text-xs font-cinzel tracking-wider border border-[#6e4620]/60 bg-[#e2ca9c]/60 text-[#381d0a] hover:bg-[#d2b278]';
          }
        });

        const meta = FOLIO_METADATA[clamped];
        folioRunningNumber.textContent = meta.num;
        folioRunningTitle.textContent = meta.title;
        folioSignatureMark.textContent = meta.signature;
        folioCatchword.textContent = meta.catchword;

        btnPrevFolio.disabled = clamped === 0;
        btnNextFolio.disabled = clamped === FOLIO_METADATA.length - 1;
        parchmentSheet.scrollTop = 0;

        if (clamped === 2) {
          drawMaterialSwatchPreview();
        }
      };

      if (!withPageTurnAnim) {
        applyDOMSwitch();
        return;
      }

      foley.playPageTurn();
      gsap.killTweensOf(pageTurnVeil);
      gsap.killTweensOf(parchmentSheet);

      const tl = gsap.timeline();
      tl.to(pageTurnVeil, { opacity: 0.72, duration: 0.18, ease: 'power2.in' }, 0);
      tl.to(parchmentSheet, {
        rotateY: -7,
        scale: 0.985,
        duration: 0.18,
        ease: 'power2.in',
        onComplete: applyDOMSwitch
      }, 0);
      tl.to(parchmentSheet, {
        rotateY: 0,
        scale: 1,
        duration: 0.28,
        ease: 'power2.out'
      }, 0.18);
      tl.to(pageTurnVeil, { opacity: 0, duration: 0.28, ease: 'power2.out' }, 0.18);
    }

    btnPrevFolio.addEventListener('click', () => {
      if (currentFolioIndex > 0) renderActiveFolio(currentFolioIndex - 1, true);
    });

    btnNextFolio.addEventListener('click', () => {
      if (currentFolioIndex < FOLIO_METADATA.length - 1) renderActiveFolio(currentFolioIndex + 1, true);

      } else if (e.key === 'Escape') {

    function drawMaterialSwatchPreview() {
      if (!swatchCanvas) return;
      const ctx = swatchCanvas.getContext('2d');
      const w = swatchCanvas.width;
      const h = swatchCanvas.height;

      const bumpVal = parseInt(sliderBump.value, 10) / 1000;
      const fireVal = parseInt(sliderFire.value, 10) / 100;
      const ageVal = parseInt(sliderAge.value, 10);

      // Update real 3D scene parameters live!
      if (pbrTextures.coverMaterial) {
        pbrTextures.coverMaterial.bumpScale = bumpVal;
      }
      fireIntensityMultiplier = fireVal;

      // Update parchment sheet aging tone live
      const lightness = Math.round(238 - ageVal * 0.32);
      const green = Math.round(218 - ageVal * 0.38);
      const blue = Math.round(180 - ageVal * 0.45);
      parchmentSheet.style.backgroundColor = `rgb(${lightness}, ${green}, ${blue})`;

      // Render interactive torchlight-on-leather preview on the 2D canvas
      const grad = ctx.createRadialGradient(w * 0.48, h * 0.42, 12, w * 0.5, h * 0.5, w * 0.72);
      const warmR = Math.min(255, Math.round(135 * fireVal));
      const warmG = Math.min(220, Math.round(72 * fireVal));
      grad.addColorStop(0, `rgb(${warmR}, ${warmG}, 28)`);
      grad.addColorStop(0.6, '#3b1d0c');
      grad.addColorStop(1, '#160903');
      ctx.fillStyle = grad;
      ctx.fillRect(0, 0, w, h);

      // Draw simulated leather relief ridges responding to bumpVal
      const ridgeContrast = Math.round(bumpVal * 1800);
      for (let i = 0; i < 420; i++) {
        const rx = (hash2D(i, 1.3) * w);
        const ry = (hash2D(i, 7.1) * h);
        ctx.fillStyle = i % 2 === 0
          ? `rgba(235, 185, 98, ${Math.min(0.55, ridgeContrast / 260)})`
          : `rgba(12, 5, 2, ${Math.min(0.65, ridgeContrast / 220)})`;
        ctx.fillRect(rx, ry, 3, 2);
      }

      ctx.strokeStyle = '#d4a249';
      ctx.lineWidth = 2;
      ctx.strokeRect(12, 12, w - 24, h - 24);

      ctx.fillStyle = '#f3e2be';
      ctx.font = 'bold 14px "Cinzel", serif';
      ctx.textAlign = 'center';
      ctx.fillText('MUESTRA DE CUERO CORDOBÁN E ILUMINACIÓN DE LUMBRE', w / 2, h / 2 - 6);
      ctx.font = 'italic 13px "EB Garamond", serif';
      ctx.fillStyle = '#dcb97a';
      ctx.fillText(`Relieve: ${(bumpVal * 100).toFixed(1)} mm · Fulgor: ${Math.round(fireVal * 100)}% · Época: ${ageVal > 60 ? 'Siglo XVI' : 'Siglo XVII'}`, w / 2, h / 2 + 18);
    }

    sliderBump.addEventListener('input', () => {

     * 13. BUCLE PRINCIPAL DE ANIMACIÓN Y FÍSICA DE LA ESCENA 3D (RENDER LOOP)
     ***************************************************************************/
    const clock = new THREE.Clock();

    function animate() {
      requestAnimationFrame(animate);
      const elapsed = clock.getElapsedTime();

      // Smooth pointer damping
      pointerNorm.x += (pointerNorm.targetX - pointerNorm.x) * 0.075;
      pointerNorm.y += (pointerNorm.targetY - pointerNorm.y) * 0.075;

      // 1. Animate volumetric campfire flames & dynamic warm light flicker
      const flicker =
        (Math.sin(elapsed * 11.3) * 0.12 +
         Math.sin(elapsed * 23.7) * 0.07 +
         Math.cos(elapsed * 6.4) * 0.09) * fireIntensityMultiplier;

      fireMainLight.intensity = (3.05 + flicker * 2.2) * fireIntensityMultiplier;
      fireMainLight.position.x = Math.sin(elapsed * 8.5) * 0.025;
      fireMainLight.position.z = 0.08 + Math.cos(elapsed * 7.2) * 0.025;
      emberBedMat.emissiveIntensity = (1.45 + flicker * 1.4) * fireIntensityMultiplier;

      flameMeshes.forEach((f) => {
        const sWave = Math.sin(elapsed * f.speed + f.phase);
        const cWave = Math.cos(elapsed * (f.speed * 0.85) + f.phase);
        f.mesh.scale.set(
          (1.0 + sWave * 0.16) * fireIntensityMultiplier,
          (1.0 + cWave * 0.24) * fireIntensityMultiplier,
          (1.0 + sWave * 0.16) * fireIntensityMultiplier
        );
        f.mesh.position.x = f.baseX + sWave * 0.022;
        f.mesh.position.z = f.baseZ + cWave * 0.022;
      });

      // 2. Animate rising glowing ember sparks
      const posArr = sparkGeo.attributes.position.array;
      for (let i = 0; i < sparkCount; i++) {
        posArr[i * 3 + 1] += sparkSpeeds[i] * 0.012;
        posArr[i * 3] += Math.sin(elapsed * 3.0 + i) * 0.0018;
        if (posArr[i * 3 + 1] > 2.2) {
          posArr[i * 3 + 1] = 0.12;
          posArr[i * 3] = (Math.random() - 0.5) * 0.42;
          posArr[i * 3 + 2] = (Math.random() - 0.5) * 0.42;
        }
      }
      sparkGeo.attributes.position.needsUpdate = true;

      // 3. Subtle pulse on the resting stump aura ring during IDLE
      auraRingMat.opacity = currentState === STATE.IDLE
        ? (isHoveringBook ? 0.52 : 0.18 + Math.sin(elapsed * 3.2) * 0.08)
        : 0.0;

      // 4. First-Person Head Movement (Fase 1 IDLE) & Held Book Inspection (Fase 2 HELD_CLOSED)
      const breathY = Math.sin(elapsed * 1.65) * 0.006;

      if (currentState === STATE.IDLE) {
        // Free head rotation looking around the campfire & toward the resting book
        camera.position.y = CAM_IDLE_POS.y + breathY;
        camera.rotation.y = CAM_LOOK_BASE.yaw - pointerNorm.x * 0.52;
        camera.rotation.x = CAM_LOOK_BASE.pitch - pointerNorm.y * 0.28;
      } else if (currentState === STATE.HELD_CLOSED) {
        // Subtle head sway + interactive 3D tilting of the closed book in the avatar's hands
        // so the user can admire the leather relief & Kevin's portrait catching the firelight!
        camera.position.y = CAM_IDLE_POS.y + breathY * 0.6;
        camera.rotation.y = CAM_LOOK_BASE.yaw - pointerNorm.x * 0.12;
        camera.rotation.x = CAM_LOOK_BASE.pitch - pointerNorm.y * 0.08;

        bookRoot.rotation.y = pointerNorm.x * 0.42;
        bookRoot.rotation.x = -0.12 + pointerNorm.y * 0.28;
        bookRoot.position.y = 1.07 + breathY * 0.8;

        leftHand.position.y = 1.04 + breathY * 0.8 - pointerNorm.x * 0.02;
        rightHand.position.y = 1.04 + breathY * 0.8 + pointerNorm.x * 0.02;
      } else if (currentState === STATE.GRABBING || currentState === STATE.OPENING_ZOOM) {
        camera.rotation.y = CAM_LOOK_BASE.yaw;
        camera.rotation.x = CAM_LOOK_BASE.pitch;
      }

      if (currentState !== STATE.READING_SINGLE_PAGE) {
        renderer.render(scene, camera);
      }
    }

    });
